cmake_minimum_required(VERSION 3.16)
project(MobiusTrajectory LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED YES)
set(CMAKE_C_EXTENSIONS NO)
add_library(mobius STATIC src/mobius.c)
target_include_directories(mobius PUBLIC include)
function(strict_c target)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wsign-conversion -Wshadow -Wundef)
  endif()
endfunction()
strict_c(mobius)
enable_testing()
add_executable(test_basis tests/test_basis.c tests/reference.c)
target_link_libraries(test_basis PRIVATE mobius)
strict_c(test_basis)
add_test(NAME basis COMMAND test_basis)
