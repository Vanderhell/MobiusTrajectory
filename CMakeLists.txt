cmake_minimum_required(VERSION 3.16)
project(MobiusTrajectory VERSION 0.1.1 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED YES)
set(CMAKE_C_EXTENSIONS NO)
add_library(mobius STATIC src/mobius.c src/mobius_inverse_table.c)
target_include_directories(mobius PUBLIC include)
function(strict_c target)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wsign-conversion -Wshadow -Wundef)
  endif()
endfunction()
strict_c(mobius)
add_executable(mobius_basic examples/basic.c)
target_link_libraries(mobius_basic PRIVATE mobius)
strict_c(mobius_basic)
enable_testing()
foreach(name basis inverse reference inverse_matrix)
  add_executable(test_${name} tests/test_${name}.c tests/reference.c)
  target_link_libraries(test_${name} PRIVATE mobius)
  strict_c(test_${name})
  add_test(NAME ${name} COMMAND test_${name})
endforeach()
